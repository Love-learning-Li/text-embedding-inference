    mask_flag = torch.ones((max_seq_len, max_seq_len),

def rotate_half(x):
    """Rotates half the hidden dims of the input."""
    x1 = x[..., : x.shape[-1] // 2]
    x2 = x[..., x.shape[-1] // 2 :]
    return torch.cat((-x2, x1), dim=-1)

def apply_rotary_pos_emb(q, k, cos, sin, unsqueeze_dim=1):
    cos = cos.unsqueeze(unsqueeze_dim)
    sin = sin.unsqueeze(unsqueeze_dim)
    q_embed = (q * cos) + (rotate_half(q) * sin)
    k_embed = (k * cos) + (rotate_half(k) * sin)
    return q_embed, k_embed

class Qwen3Attention:
    def forward(self, hidden_states, position_embeddings, cu_seqlens, max_s, attn_mask=None):
        if hidden_states.dim() != 2:
            raise ValueError(f"[Qwen3Attention] TND required: hidden_states must be 2D [T,H], got {tuple(hidden_states.shape)}")

        input_shape = hidden_states.shape[:-1]
        q_proj_out = F.linear(hidden_states, self.q_proj_weight)
        q = self.q_norm.forward(
            q_proj_out.view(*input_shape, self.num_heads, self.head_dim)
        )

        k_proj_out = F.linear(hidden_states, self.k_proj_weight)
        k = self.k_norm.forward(
            k_proj_out.view(*input_shape, self.num_key_value_heads, self.head_dim)
        )

        v = F.linear(hidden_states, self.v_proj_weight).view(
            *input_shape, self.num_key_value_heads, self.head_dim
        )

        cos, sin = position_embeddings
        if self.layer_idx == 0:
            log_shape_once(...)

        q, k = apply_rotary_pos_emb(q, k, cos, sin, unsqueeze_dim=1)

        if self.num_key_value_groups > 1:
            k = k.repeat_interleave(self.num_key_value_groups, dim=1)
            v = v.repeat_interleave(self.num_key_value_groups, dim=1)

        attn_output = torch.empty_like(q)
        attention(
            q, k, v, self.num_heads, attn_output, cu_seqlens, max_s,
            self.softmax_scale, is_causal=True, attn_mask=attn_mask,
        )

        attn_output = attn_output.reshape(*input_shape, -1).contiguous()
        attn_output = F.linear(attn_output, self.o_proj_weight, bias=None)
        return attn_output
```

**修改目的**：
- 适配 TND 格式（Token-Numhead-Dim）的输入输出
- 支持 GQA（Grouped Query Attention）的 key/value 头复制
- 添加维度检查和日志记录

**修改原因**：
- 原始代码假设输入为 BNSD 格式（Batch-Numhead-Seq-Dim），需要适配 TND 格式
- NPU 的 `npu_fusion_attention` 期望 TND 格式输入
- 需要处理 Qwen3 的 GQA 机制（num_key_value_heads < num_heads）

**修改思路**：
- 添加严格的维度检查，确保输入为 2D（实际是展平后的 TND）
- 修改 view 操作的维度参数，从 `(-1, head_dim)` 改为 `(num_heads, head_dim)`
- RoPE 应用改为 `unsqueeze_dim=1` 以适应 TND
- 添加 GQA 支持：使用 `repeat_interleave` 复制 key/value 头
- 传递 `self.num_heads` 给 attention 函数

        attn_mask = None
        max_input_lens = batch.max_s

        hidden_states = output.last_hidden_state

            last_indices = (cu_seqlens[1:] - 1).to(device=hidden_states.device, dtype=torch.long)

            pooling_attention_mask[i, :seq_len] = 1
            offset += seq_len

def _get_npu_compressed_causal_mask(device: torch.device) -> torch.Tensor:
    """生成并缓存 NPU 压缩因果掩码"""
    device_key = f"{device.type}:{device.index}"
    cached = _NPU_COMPRESSED_MASK_CACHE.get(device_key)
    if cached is not None:
        return cached

    mask = torch.triu(
        torch.ones((NPU_COMPRESSED_MASK_SIZE, NPU_COMPRESSED_MASK_SIZE), dtype=torch.bool, device=device),
        diagonal=1,
    )
    _NPU_COMPRESSED_MASK_CACHE[device_key] = mask
    return mask

def _npu_warn_once(key: str, message: str):
    """NPU 警告日志（仅输出一次）"""
    if key in _NPU_WARN_ONCE_KEYS:
        return
    _NPU_WARN_ONCE_KEYS.add(key)
    logger.warning(message)

def _eager_varlen_tnd_attn(
    q: torch.Tensor, k: torch.Tensor, v: torch.Tensor,
    actual_seq_qlen: list, softmax_scale: float, is_causal: bool,
) -> torch.Tensor:
    """Eager 模式的变长序列注意力实现（fallback）"""
    out = torch.empty_like(q)
    start = 0
    for end in actual_seq_qlen:
        seq_end = int(end)
        if seq_end <= start:
            continue

        q_i = q[start:seq_end].transpose(0, 1)  # [H, S, D]
        k_i = k[start:seq_end].transpose(0, 1)
        v_i = v[start:seq_end].transpose(0, 1)

        scores = torch.matmul(q_i, k_i.transpose(-1, -2)) * softmax_scale
        if is_causal:
            s = seq_end - start
            causal = torch.triu(torch.ones((s, s), dtype=torch.bool, device=q.device), diagonal=1)
            scores = scores.masked_fill(causal.unsqueeze(0), float("-inf"))

        probs = torch.softmax(scores, dim=-1, dtype=torch.float32).to(q.dtype)
        out_i = torch.matmul(probs, v_i).transpose(0, 1).contiguous()  # [S, H, D]
        out[start:seq_end].copy_(out_i)
        start = seq_end

    return out
```

**修改目的**：
- `_get_npu_compressed_causal_mask`：缓存因果掩码避免重复生成
- `_npu_warn_once`：防止 NPU 相关警告刷屏
- `_eager_varlen_tnd_attn`：当 NPU 融合算子失败时的 fallback 实现

## 5. 跨文件修改总结

### 5.1 TND 格式改造（核心架构改动）

**涉及文件**：`flash_qwen3.py`, `flash_attn.py`, `types.py`

**改动内容**：
- **数据格式**：从 BNSD（Batch-Numhead-Seq-Dim）改为 TND（Token-Numhead-Dim）
- **变长序列支持**：通过 `cu_seqlens`（累积序列长度）处理变长 batch
- **维度处理**：所有张量操作适配 3D TND 格式

**设计考量**：
1. **显存效率**：TND 避免了对齐 padding 的显存浪费，适合变长序列推理
2. **算子优化**：NPU 的 `npu_fusion_attention` 对 TND 格式有专门优化
3. **复杂度权衡**：TND 增加了代码复杂度，但提升了长序列场景的性能

### 5.2 NPU 算子替换策略

**涉及文件**：`flash_qwen3.py`, `flash_attn.py`

| 原始实现 | NPU 算子 | 说明 |

| 手动 Attention | `torch_npu.npu_fusion_attention` | 融合注意力算子 |

**Fallback 机制**：
- 所有 NPU 算子都有 Python fallback 实现
- 使用全局变量记录算子可用性，失败时自动降级
- 保证在算子不支持时的可用性

### 5.3 模型加载双模式支持

**涉及文件**：`flash_qwen3.py`

| 注意力输出异常 | TND 格式维度处理错误 | 添加严格的维度检查 |
| 单文件模型加载失败 | 强制读取 index.json | 增加文件存在性检测 |
| 性能低于预期 | 调试日志开销 | 添加日志开关，默认关闭 |

### 6.2 NPU 适配关键点

1. **设备类型**：将 `autocast` 的默认设备从 "cpu" 改为 "npu"
2. **张量放置**：`cu_seqlens` 保留在 CPU 上，避免 NPU 算子参数问题
3. **算子限制**：NPU 算子对输入维度、格式有严格要求，需要详细测试
4. **Fallback 必备**：NPU 算子可能在某些场景下失败，必须有降级方案
